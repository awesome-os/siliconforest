# Only the reviewed observer is compiled here. The unchanged ten-TU core is
# linked exactly once from the separately reviewed cached object provider.
add_kext_bundle(PDBCMPrivateProbe.kext INFO_PLIST Info.plist KERNEL_PRIVATE
    BUNDLE_IDENTIFIER org.darwinlab.driver.PDBCMPrivateProbe BUNDLE_VERSION 0.1.0
    ANTIMAIN_FUNCTION pdbcm_private_stop)
set_property(TARGET PDBCMPrivateProbe.kext PROPERTY OUTPUT_NAME PDBCMPrivateProbe)
target_sources(PDBCMPrivateProbe.kext PRIVATE
    observer/PDBCMPrivateProbe.cpp observer/probe.c observer/protocol.c
    private_policy_provider.o)
set_source_files_properties(private_policy_provider.o PROPERTIES EXTERNAL_OBJECT TRUE)
target_include_directories(PDBCMPrivateProbe.kext PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/observer)
target_compile_options(PDBCMPrivateProbe.kext PRIVATE
    -Wall -Wextra -Wframe-larger-than=1024 -Werror=frame-larger-than
    -ffreestanding -fno-builtin -fstack-usage
    $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>
    $<$<COMPILE_LANGUAGE:CXX>:-fno-rtti>)
set_property(TARGET PDBCMPrivateProbe.kext PROPERTY C_STANDARD 11)
set_property(TARGET PDBCMPrivateProbe.kext PROPERTY CXX_STANDARD 17)
install(TARGETS PDBCMPrivateProbe.kext DESTINATION System/Library/Extensions COMPONENT BaseSystem)
